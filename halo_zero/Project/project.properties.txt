[all]
//The author of this game
//Currently only for WHDLoad
author=Earok

//Three letter month such as JAN or SEP, used in the Mega Drive copyright string.
copyright_month=Dec

//Four digit year of release.
//Used in the Mega Drive copyright string, the NeoSD header and the Neo Geo MAME entry.
copyright_year=2025

//How many decals the Level->Decal command may paint into the level in a single frame, up to 254.
//Zero disables decals entirely and costs no memory. Going over the limit is a kill screen.
//Amiga and Mega Drive paint decals, NeoGeo does not yet.
decals=16

//A longer description of the game.
//At present only the Neo Geo MAME entry uses it.
description=Halo Zero

//Set true to skip the sanity checks on actors, such as a platformer actor whose collision box starts below the ground line.
experiment_ignore_actor_configuration_errors=false

//Set true to list Scorpion's own standard library codeblocks in the code editor.
//They are hidden normally because changing them can break the engine.
experiment_show_internal_codeblocks=true

//The name of the game.
//Written into the Mega Drive ROM header and used as the name of the save file.
project_name=Halo Zero

//Overseas title for the Mega Drive ROM header.
//Leave blank to reuse project_name.
project_name_international=Halo Zero

//Who published the game.
//The Mega Drive copyright string only keeps the first four characters.
publisher=Earok

//Version number reported by the WHDLoad slave. Amiga only.
version=1

[amiga_all]
//Atlas packing setting, it's difficult to quantify the amount of saving but feel free to experiment with this value to see if memory usage goes up or down.
//Zero turns atlas packing off.
atlas_compression=0

//Set true to use a custom panel gap size, overriding Scorpion's bare minimum.
//If the gap is too low, graphical corruption will result.
experiment_allow_any_size_panel_gap=false

//Number of screen buffers the engine allocates, where two means double buffering.
//Only 1 or 2 buffers currently supported
experiment_buffer_count=2

//Set true to build without a double buffered copper list.
//Saves memory at a high risk of graphical corruption when loading or unloading panels.
experiment_disable_double_copperlist=false

//Set true to skip the check on level width and height.
//Levels past the limit may not display properly.
experiment_exceed_max_map_size=false

//Lets exported graphics overlap bytes that already exist in the bundle, which saves chip memory but makes compiling slower and introduces a slight risk of graphical glitches.
tightly_pack_graphics=true

[megadrive]
//Set true to skip the panel height checks that normally cap panels at 224 lines on NTSC and 240 on PAL.
experiment_allow_any_size_panel=false

//Set true to allow compiling with the sound driver set to None, which can result in instability.
experiment_allow_no_audio_driver=false

//Two digit revision that follows the serial number in the ROM header.
sega_revision=00

//Eight digit serial for the ROM header.
//A hash of the project name is used when this is blank.
sega_serial_number=

//Two letter software type from the Sega spec, GM being a game.
//Written to the front of the serial number in the ROM header.
sega_software_type=GM

//Console name written to the start of the ROM header.
//Change it only for hardware that expects something else, such as SEGA 32X or SEGA PICO.
sega_system=SEGA MEGA DRIVE

[megadrive_or_neogeo]
//Set true to carry on building when accounted RAM goes over 44KB.
//The game may run out of memory.
experiment_ignore_memory_usage_error=false

[neogeo]
//Genre shown in the NeoSD menu.
//One of Other, Action, BeatEmUp, Sports, Driving, Platformer, Mahjong, Shooter, Quiz, Fighting or Puzzle.
neogeo_genre=Other

//NGH number in hex, which is how the BIOS tells one cart from another.
//Falls back to 5345 when blank, so pick something unused for a hardware release.
neogeo_ngh_hex=

//Use a single VROM for both ADPCM-A samples and ADPCM-B music.
//Turning this off allows for more samples to be stored at the cost of additional publishing complexity and manufacture cost.
neogeo_single_voice_rom=true

[whdload_13]
//Chip memory the WHDLoad slave asks for, in hex. $100000 is one megabyte.
//WHDLoad will never request more ChipRAM than this so even if a user has a lot of ChipRAM, it may crash with out of memory if you don't request enough.
chip_mem_size=$110000

//Fast memory the WHDLoad slave asks for, in hex. $0 asks for none.
//WHDLoad will never request more FastRAM than this so even if a user has a lot of FastRAM, it may crash with out of memory if you don't request enough.
fast_mem_size=$100000

[whdload_31]
//Chip memory the WHDLoad slave asks for, in hex. $100000 is one megabyte.
//WHDLoad will never request more ChipRAM than this so even if a user has a lot of ChipRAM, it may crash with out of memory if you don't request enough.
chip_mem_size=$1ff000

//Fast memory the WHDLoad slave asks for, in hex. $0 asks for none.
//WHDLoad will never request more FastRAM than this so even if a user has a lot of FastRAM, it may crash with out of memory if you don't request enough.
fast_mem_size=$0
